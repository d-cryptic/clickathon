  SELECT toDateTime(m) AS minute, uniqExact(video_session_id) c
  FROM (SELECT video_session_id,
          arrayJoin(range(toUInt32(toStartOfMinute(interval_start)),
                          toUInt32(toStartOfMinute(interval_end))+1, 60)) AS m
        FROM bg_state_gate.si_M)
  GROUP BY minute ORDER BY c DESC, minute ASC LIMIT 1;
