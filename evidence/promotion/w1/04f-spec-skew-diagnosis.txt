== Appendix: the same-second verification query (read-only)

WITH p AS (SELECT video_session_id, toUInt32(event_timestamp) AS t FROM ev_raw WHERE event = 'pause'),
     r AS (SELECT video_session_id, toUInt32(event_timestamp) AS t FROM ev_raw WHERE event = 'resume')
SELECT
  (SELECT count() FROM p) AS total_pauses,
  countIf(hit) AS same_second_resume,
  round(100.0 * countIf(hit) / (SELECT count() FROM p), 2) AS pct
FROM (SELECT p.video_session_id, p.t, max(r.t = p.t) AS hit
      FROM p LEFT JOIN r ON r.video_session_id = p.video_session_id
      GROUP BY p.video_session_id, p.t)

-> 27340 · 2502 · 9.15
