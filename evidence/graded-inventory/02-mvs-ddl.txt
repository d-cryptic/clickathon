name:               mv_user_minute
create_table_query: CREATE MATERIALIZED VIEW sonyliv.mv_user_minute TO sonyliv.cc_user_minute (`minute` DateTime, `platform` LowCardinality(String), `country` LowCardinality(String), `content_id` Int64, `active_state` AggregateFunction(uniqExact, String)) AS SELECT toDateTime(m) AS minute, platform, country, content_id, uniqExactState(user_id) AS active_state FROM (SELECT user_id, platform, country, content_id, arrayJoin(range(toUInt32(toStartOfMinute(interval_start)), toUInt32(toStartOfMinute(interval_end)) + 1, 60)) AS m FROM sonyliv.session_intervals) GROUP BY minute, platform, country, content_id
