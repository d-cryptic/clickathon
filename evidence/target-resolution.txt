# evidence/target-resolution.txt — one command, one database (ADR 0018)
# Captured live on 2026-08-01 against TARGET=local (docker 'ch', ClickHouse 26.7.1.1315)
# and TARGET=cloud (READ-ONLY queries only — the graded service was never written to).
# Regenerate: the commands are inline below; every output block is verbatim.

## 1. Same command, both targets — each resolves to exactly one database

$ Q="SELECT currentDatabase() AS db, count() AS stateless_rows, (SELECT max(c) FROM (SELECT minute, uniqExactMerge(active_state) AS c FROM cc_minute_stateless GROUP BY minute)) AS peak FROM cc_minute_stateless"
$ tools/ch "$Q"          # TARGET=local
db	stateless_rows	peak
default	91292	2894
$ tools/ch -c "$Q"       # TARGET=cloud (read)
db	stateless_rows	peak
sonyliv	91292	2894

# Identical numbers are the success condition: same 12-day file on both sides,
# and — after the drift fix below — the same uniqExact schema. Before the fix the
# local table could not even execute uniqExactMerge (its state was uniq/HLL).
